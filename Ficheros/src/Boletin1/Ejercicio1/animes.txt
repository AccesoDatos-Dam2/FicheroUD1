package Boletin1.Ejercicio1;

import java.io.File;
import java.io.FileNotFoundException;
import java.util.Scanner;

public class fichero {
    public static void main(String[] args){
        File f1=new File("/home/alumno/IdeaProjects/githubAccesoDatos/Ficheros/src/Boletin1/Ejercicio1/personajes.txt");
        File f2=new File("/home/alumno/IdeaProjects/githubAccesoDatos/Ficheros/src/Boletin1/Ejercicio1/animes.txt");
        try{
            Scanner sc1=new Scanner(f2);

            while(sc1.hasNextLine()){
                String linea1=sc1.nextLine().trim();
                if(linea1.isEmpty()){
                    continue;
                }
                int espacioSerie=linea1.indexOf(" ");
                String idSerie=linea1.substring(0,espacioSerie);
                String nombreSerie=linea1.substring(espacioSerie+1);
                System.out.println(nombreSerie);

                Scanner sc2= new Scanner(f1);
                boolean encontroPersonaje=false;

                while(sc2.hasNextLine()){
                    String linea2=sc2.nextLine().trim();
                    if(linea2.isEmpty()){
                        continue;
                    }
                    int espacioPersonaje=linea2.indexOf(" ");
                    String idPersonaje=linea2.substring(0,espacioPersonaje);
                    String nombrePersonaje=linea2.substring(espacioPersonaje+1);

                    if(idSerie.equals(idPersonaje)){
                        System.out.println("- "+nombrePersonaje);
                        encontroPersonaje=true;
                    }
                }
                sc2.close();
                if(!encontroPersonaje){
                    System.out.println("- No hay personajes");
                }
            }
            sc1.close();
        }catch (FileNotFoundException e){
            System.out.println("Error: No se encontro alguno de los ficheros"+e.getMessage());
        }
    }
}
